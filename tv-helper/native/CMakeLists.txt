cmake_minimum_required(VERSION 3.24)
project(lgxmb_home_hook C)
set(CMAKE_C_STANDARD 99)
find_package(ezinject REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GUM REQUIRED frida-gum-1.0)
if(NOT HOME_HOOK_BUILD_ID MATCHES "^[0-9a-f]+$")
    message(FATAL_ERROR "Build through tools/build-home-hook.sh to provide the source identity")
endif()
add_ezinject_library(lgxmb-home-hook SOURCES home-hook.c)
set_target_properties(lgxmb-home-hook PROPERTIES PREFIX "")
target_compile_definitions(lgxmb-home-hook PRIVATE HOME_HOOK_BUILD_ID="${HOME_HOOK_BUILD_ID}")
target_compile_options(lgxmb-home-hook PRIVATE -Wall -Wextra -Werror -ffunction-sections -fdata-sections)
target_include_directories(lgxmb-home-hook PRIVATE ${GUM_INCLUDE_DIRS})
# Bundle Gum's dependencies; use only the TV's standard C/POSIX runtime libraries.
set(bundled_libraries ${GUM_LIBRARIES} ffi pcre2-8 dwarf)
foreach(name IN LISTS bundled_libraries)
    unset(archive CACHE)
    find_library(archive NAMES "lib${name}.a" REQUIRED)
    target_link_libraries(lgxmb-home-hook "${archive}")
endforeach()
target_link_libraries(lgxmb-home-hook dl pthread rt m)
target_link_options(lgxmb-home-hook PRIVATE -Wl,--gc-sections -Wl,--exclude-libs,ALL -Wl,--no-undefined)
